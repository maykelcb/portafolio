-- Consulta de rendimiento analítico
SELECT 
    kpi.NombreIndicador,
    SUM(met.ValorReal) AS MetricaTotal,
    ROUND(AVG(met.Eficiencia), 2) AS RatioCumplimiento
FROM Fact_Operaciones met
INNER JOIN Dim_Indicadores kpi ON met.IndicadorID = kpi.ID
WHERE met.Fecha >= '2026-01-01'
GROUP BY kpi.NombreIndicador
ORDER BY MetricaTotal DESC;
